{{- /* Proposition numeral of a post: its chronological rank among all posts, as a roman numeral. Posts sharing a date are ordered by file path, so numerals are stable and distinct. Usage: partial "numeral.html" .Page → "XI" */ -}} {{- $page := . -}} {{- $sorted := (sort (where site.RegularPages "Section" "posts") "File.Path").ByDate -}} {{- $rank := 0 -}} {{- range $i, $p := $sorted -}} {{- if eq $p.File.Path $page.File.Path }}{{ $rank = add $i 1 }}{{ end -}} {{- end -}} {{- $values := slice 1000 900 500 400 100 90 50 40 10 9 5 4 1 -}} {{- $symbols := slice "M" "CM" "D" "CD" "C" "XC" "L" "XL" "X" "IX" "V" "IV" "I" -}} {{- $out := "" -}} {{- range $k, $v := $values -}} {{- range seq 1 12 -}} {{- if ge $rank $v }}{{ $out = printf "%s%s" $out (index $symbols $k) }}{{ $rank = sub $rank $v }}{{ end -}} {{- end -}} {{- end -}} {{- return $out -}}

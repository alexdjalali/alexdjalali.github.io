{{- /* Proposition numeral of a post: its chronological rank among all posts, as a roman numeral. Posts sharing a date are ordered by file path, so numerals are stable and distinct. Usage: partial "numeral.html" .Page → "XI" */ -}} {{- $page := . -}} {{- $sorted := (sort (where site.RegularPages "Section" "posts") "File.Path").ByDate -}} {{- $rank := 0 -}} {{- range $i, $p := $sorted -}} {{- if eq $p.File.Path $page.File.Path }}{{ $rank = add $i 1 }}{{ end -}} {{- end -}} {{- return partial "roman.html" $rank -}}
